{{- $img := .Page.Resources.GetMatch .Destination -}} {{- $class := .Attributes.class | default "feature" -}} {{- if not $img -}} {{- $img = resources.Get .Destination -}} {{- end -}} {{- if $img -}} {{- $mediaType := $img.MediaType.SubType -}} {{- $desktopWidth := "1280x" -}} {{- $sizesAttr := "(max-width: 480px) 100vw, (max-width: 800px) 90vw, 1280px" -}} {{- if eq $class "full" -}} {{- $desktopWidth = "1920x" -}} {{- $sizesAttr = "(max-width: 480px) 100vw, (max-width: 800px) 90vw, 1920px" -}} {{- end -}} {{- if or (eq $mediaType "svg+xml") (eq $mediaType "svg") (eq $mediaType "gif") (eq $mediaType "avif") -}} {{/* WebP: only resize */}} {{- else if eq $mediaType "webp" -}} {{- $mobile := $img.Resize "600x" -}} {{- $tablet := $img.Resize "800x" -}} {{- $desktop := $img.Resize $desktopWidth -}} {{/* PNG and JPEG: convert to AVIF with 80% quality and resize */}} {{- else -}} {{- $mobileWebp := $img.Resize "600x webp q85 text catmullRom" -}} {{- $tabletWebp := $img.Resize "800x webp q85 text catmullRom" -}} {{- $desktopWebp := $img.Resize (printf "%s webp q85 text lanczos" $desktopWidth) -}} {{- $mobile := $img.Resize "600x" -}} {{- $tablet := $img.Resize "800x" -}} {{- $desktop := $img.Resize $desktopWidth -}} {{- end -}} {{- else -}} {{/* Fallback for external images */}} {{- end -}}
